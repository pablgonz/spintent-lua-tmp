% Problema: con DOS \spinterval en la misma fórmula, el segundo no se lee
% (test-019, casos 68, 69, 70 y 73: «... unión», «... intersección»). Con
% uno solo funciona siempre. Cada caso cambia UNA sola cosa respecto del
% control V0a; anotar para cada uno SI SE OYE el segundo intervalo
% («intervalo abierto en 2 y cerrado en 6»). Los comentarios "doc:" son
% lo que se oiría si se leyera completo.
%
% V0a usa el comando real. V0b lo reproduce a mano con los nativos de luamml
% (\MathMLintent y \MathMLarg) y produce exactamente el mismo MathML (se
% verificó en el XML real); V1 a V8 son variantes de V0b:
%   V1  nombres de arg unicos en el segundo intervalo (ab2, a2, b2)
%   V2  sin el separador invisible entre la interseccion y el segundo
%   V3  sin el separador invisible antes del primero
%   V4  sin ninguno de los dos separadores
%   V5  el segundo intervalo se reemplaza por x
%   V6  el primer intervalo se reemplaza por x
%   V7  igual que V0b, con el signo igual en lugar de la interseccion
%   V8  cada extremo numerico va envuelto con \__spintent_inner_sep:
%
% Observacion del XML: \spnum solo emite su separador en el PRIMER numero de
% la fórmula, asi que el segundo intervalo queda con <mn> , <mn> pelados
% (el primero lleva <mrow><mi sep/><mn>). V8 lo corrige sin tocar \spnum.
%
% Si se oye en V1 -> los nombres repetidos; en V2 o V3 -> los separadores;
% en V5, V6 o V7 -> la combinacion de dos intervalos o el operador; en V8
% -> la adyacencia de los numeros, y ese es el arreglo.

% Asistentes que reproducen la estructura que arma \spinterval.
\ExplSyntaxOn
\NewDocumentCommand \expivl { m m m m m m m m m }
  {
    \MathMLintent { #1 }
      {
        \MathMLintent { :silent } { #6 }
        \MathMLarg { #3 }
          {
            \__spintent_inner_sep:
            \MathMLintent { #2 }
              {
                \MathMLarg { #4 } { \spnum { #8 } }
                \MathMLintent { :silent } { , }
                \MathMLarg { #5 } { \spnum { #9 } }
              }
          }
        \MathMLintent { :silent } { #7 }
      }
  }
\NewDocumentCommand \expsep { }
  { \MathMLintent { :silent } { \c__spintent_invisible_sep_int } }
\NewDocumentCommand \expwone { }
  {
    \expivl
      { _intervalo-cerrado-de:prefix($ab) } { _a:infix($a,$b) } { ab } { a } { b }
      { \mathopen { \lbrack } } { \mathclose { \rbrack } } { 1 } { 4 }
  }
\NewDocumentCommand \expwtwo { }
  {
    \expivl
      { _intervalo-abierto-en:prefix($ab) } { _y-cerrado-en:infix($a,$b) } { ab } { a } { b }
      { \mathopen { \rbrack } } { \mathclose { \rbrack } } { 2 } { 6 }
  }
\NewDocumentCommand \expwtwob { }
  {
    \expivl
      { _intervalo-abierto-en:prefix($ab2) } { _y-cerrado-en:infix($a2,$b2) } { ab2 } { a2 } { b2 }
      { \mathopen { \rbrack } } { \mathclose { \rbrack } } { 2 } { 6 }
  }
\NewDocumentCommand \expivlw { m m m m m m m m m }
  {
    \MathMLintent { #1 }
      {
        \MathMLintent { :silent } { #6 }
        \MathMLarg { #3 }
          {
            \__spintent_inner_sep:
            \MathMLintent { #2 }
              {
                \MathMLarg { #4 } { \__spintent_inner_sep: \spnum { #8 } }
                \MathMLintent { :silent } { , }
                \MathMLarg { #5 } { \__spintent_inner_sep: \spnum { #9 } }
              }
          }
        \MathMLintent { :silent } { #7 }
      }
  }
\NewDocumentCommand \expwonew { }
  {
    \expivlw
      { _intervalo-cerrado-de:prefix($ab) } { _a:infix($a,$b) } { ab } { a } { b }
      { \mathopen { \lbrack } } { \mathclose { \rbrack } } { 1 } { 4 }
  }
\NewDocumentCommand \expwtwow { }
  {
    \expivlw
      { _intervalo-abierto-en:prefix($ab) } { _y-cerrado-en:infix($a,$b) } { ab } { a } { b }
      { \mathopen { \rbrack } } { \mathclose { \rbrack } } { 2 } { 6 }
  }
\ExplSyntaxOff
\begin{document}

% doc: «intervalo cerrado de 1 a 4 intersección intervalo abierto en 2 y cerrado en 6»
1. V0a. Control con el comando real: $\spinterval{[1,4]} \cap \spinterval{]2,6]}$.

% doc: «intervalo cerrado de 1 a 4 intersección intervalo abierto en 2 y cerrado en 6»
2. V0b. El mismo control hecho a mano con los nativos (debe oírse igual que V0a): $\expsep \expwone \cap \expsep \expwtwo$.

% doc: «intervalo cerrado de 1 a 4 intersección intervalo abierto en 2 y cerrado en 6»
3. V1. Nombres de argumento únicos en el segundo intervalo: $\expsep \expwone \cap \expsep \expwtwob$.

% doc: «intervalo cerrado de 1 a 4 intersección intervalo abierto en 2 y cerrado en 6»
4. V2. Sin el separador invisible entre la intersección y el segundo intervalo: $\expsep \expwone \cap \expwtwo$.

% doc: «intervalo cerrado de 1 a 4 intersección intervalo abierto en 2 y cerrado en 6»
5. V3. Sin el separador invisible antes del primer intervalo: $\expwone \cap \expsep \expwtwo$.

% doc: «intervalo cerrado de 1 a 4 intersección intervalo abierto en 2 y cerrado en 6»
6. V4. Sin ninguno de los dos separadores: $\expwone \cap \expwtwo$.

% doc: «intervalo cerrado de 1 a 4 intersección x»
7. V5. El segundo intervalo reemplazado por x: $\expsep \expwone \cap \expsep x$.

% doc: «x intersección intervalo abierto en 2 y cerrado en 6»
8. V6. El primer intervalo reemplazado por x: $\expsep x \cap \expsep \expwtwo$.

% doc: «intervalo cerrado de 1 a 4 es igual a intervalo abierto en 2 y cerrado en 6»
9. V7. Con el signo igual en lugar de la intersección: $\expsep \expwone = \expsep \expwtwo$.

% doc: «intervalo cerrado de 1 a 4 intersección intervalo abierto en 2 y cerrado en 6»
10. V8. Los extremos numéricos siempre envueltos con el separador interno (el arreglo candidato): $\expsep \expwonew \cap \expsep \expwtwow$.

\end{document}
